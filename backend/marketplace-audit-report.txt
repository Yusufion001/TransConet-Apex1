     6	async function expireMarketplaceLifecycle() {
     7	  const now = new Date();
     8	
     9	  return prisma.$transaction(async (tx) => {
    10	    const expiredBids = await tx.marketplaceBid.findMany({
    11	      where: {
    12	        status: "PENDING",
    13	        expiresAt: {
    14	          not: null,
    15	          lte: now,
    16	        },
    17	      },
    18	      select: {
    19	        id: true,
    20	        requestId: true,
    21	        transporterId: true,
    22	      },
    23	    });
    24	
    25	    if (expiredBids.length > 0) {
    26	      await tx.marketplaceBid.updateMany({
    27	        where: {
    28	          id: {
    29	            in: expiredBids.map((bid) => bid.id),
    30	          },
    31	          status: "PENDING",
    32	          expiresAt: {
    33	            not: null,
    34	            lte: now,
    35	          },
    36	        },
    37	        data: {
    38	          status: "EXPIRED",
    39	        },
    40	      });
    41	    }
    42	
    43	    /*
    44	     * MarketplaceRequest.scheduledDate represents the planned
    45	     * shipment/pickup date. It is NOT the marketplace request's
    46	     * expiration timestamp.
    47	     *
    48	     * Therefore a scheduled load must remain OPEN until:
    49	     * - a bid is selected, or
    50	     * - the request is explicitly closed by another lifecycle action.
    51	     *
    52	     * Only MarketplaceBid.expiresAt controls bid expiration.
    53	     */
    54	
    55	    return {
    56	      expiredBidIds: expiredBids.map((bid) => bid.id),
    57	    };
    58	  });
    59	}

   283	export async function createMarketplaceBid(data: {
   284	  requestId: string;
   285	  transporterId: string;
   286	  vehicleId: string;
   287	  amount: number;
   288	  message?: string;
   289	  expiresAt?: Date;
   290	}) {
   291	  await expireMarketplaceLifecycle();
   292	
   293	  const request = await prisma.marketplaceRequest.findUnique({
   294	    where: { id: data.requestId },
   295	    select: {
   296	      id: true,
   297	      status: true,
   298	      truckCategory: true,
   299	      scheduledDate: true,
   300	      customerId: true,
   301	    },
   302	  });
   303	
   304	  if (!request) {
   305	    throw new Error("Marketplace request not found");
   306	  }
   307	
   308	  if (request.status !== "OPEN") {
   309	    throw new Error("Marketplace request is no longer accepting bids");
   310	  }
   311	
   312	  if (request.customerId === data.transporterId) {
   313	    throw new Error("Customer cannot bid on their own marketplace request");
   314	  }
   315	
   316	  const transporter = await prisma.user.findUnique({
   317	    where: { id: data.transporterId },
   318	    select: {
   319	      id: true,
   320	      role: true,
   321	      status: true,
   322	      transporterProfile: {
   323	        select: {
   324	          verificationStatus: true,
   325	        },
   326	      },
   327	    },
   328	  });
   329	
   330	  if (!transporter || transporter.role !== "TRANSPORTER") {
   331	    throw new Error("Only transporters can submit bids");
   332	  }
   333	
   334	  if (transporter.status !== "ACTIVE") {
   335	    throw new Error("Transporter account is not active");
   336	  }
   337	
   338	  if (
   339	    !transporter.transporterProfile ||
   340	    transporter.transporterProfile.verificationStatus !== "APPROVED"
   341	  ) {
   342	    throw new Error("Transporter is not approved");
   343	  }
   344	
   345	  const vehicle = await prisma.vehicle.findUnique({
   346	    where: { id: data.vehicleId },
   347	    select: {
   348	      id: true,
   349	      transporterId: true,
   350	      vehicleType: true,
   351	      vehicleClass: true,
   352	      verificationStatus: true,
   353	      availabilityStatus: true,
   354	    },
   355	  });
   356	
   357	  if (!vehicle) {
   358	    throw new Error("Vehicle not found");
   359	  }
   360	
   361	  if (vehicle.transporterId !== data.transporterId) {
   362	    throw new Error("Vehicle does not belong to transporter");
   363	  }
   364	
   365	  if (vehicle.verificationStatus !== "APPROVED") {
   366	    throw new Error("Vehicle is not approved");
   367	  }
   368	
   369	  if (vehicle.availabilityStatus !== "AVAILABLE") {
   370	    throw new Error("Vehicle is not available");
   371	  }
   372	
   373	  if (
   374	    request.truckCategory &&
   375	    vehicle.vehicleType !== request.truckCategory &&
   376	    vehicle.vehicleClass !== request.truckCategory
   377	  ) {
   378	    throw new Error("Vehicle does not match requested truck category");
   379	  }
   380	
   381	  if (data.expiresAt && data.expiresAt <= new Date()) {
   382	    throw new Error("Bid expiration must be in the future");
   383	  }
   384	
   385	  try {
   386	    const bid = await prisma.marketplaceBid.create({
   387	      data: {
   388	        requestId: data.requestId,
   389	        transporterId: data.transporterId,
   390	        vehicleId: data.vehicleId,
   391	        amount: data.amount,
   392	        message: data.message,
   393	        expiresAt: data.expiresAt,
   394	        status: "PENDING",
   395	      },
   396	      include: {
   397	        vehicle: true,
   398	      },
   399	    });
   400	
   401	    publishEvent("marketplace", {
   402	      eventType: "BID_SUBMITTED",
   403	      module: "FLEET_MARKETPLACE",
   404	      entityType: "MARKETPLACE_BID",
   405	      entityId: bid.id,
   406	      actorId: data.transporterId,
   407	      data: {
   408	        bidId: bid.id,
   409	        requestId: data.requestId,
   410	        transporterId: data.transporterId,
   411	        vehicleId: data.vehicleId,
   412	        amount: bid.amount,
   413	        status: bid.status,
   414	      },
   415	    });
   416	
   417	    return bid;
   418	  } catch (error) {
   419	    if (
   420	      error instanceof Error &&
   421	      error.message.includes("Unique constraint")
   422	    ) {
   423	      throw new Error("Transporter has already submitted a bid for this request");
   424	    }
   425	
   426	    throw error;
   427	  }
   428	}

   436	    where: { id: bidId },
   437	    select: {
   438	      id: true,
   439	      requestId: true,
   440	      transporterId: true,

   444	
   445	  if (!bid) {

   465	    eventType: "BID_WITHDRAWN",
   466	    module: "FLEET_MARKETPLACE",
   467	    entityType: "MARKETPLACE_BID",
   468	    entityId: updated.id,
   469	    actorId: transporterId,
   470	    data: {
   471	      bidId: updated.id,
   472	      requestId: updated.requestId,
   473	      status: updated.status,
   474	    },
   475	  });

   481	  requestId: string,
   482	  bidId: string,
   483	  customerId: string,
   484	) {
   485	  await expireMarketplaceLifecycle();
   486	  const result = await prisma.$transaction(async (tx) => {
   487	    /*
   488	     * Read the marketplace request and verify ownership.
   489	     * The request must still be OPEN when the selection begins.
   490	     */
   491	    const request = await tx.marketplaceRequest.findUnique({
   492	      where: {
   493	        id: requestId,
   494	      },
   495	      select: {
   496	        id: true,
   497	        customerId: true,
   498	        status: true,
   499	        cargoDescription: true,
   500	        truckCategory: true,
   501	        cargoCategory: true,
   502	        cargoWeight: true,
   503	        pickupLocation: true,
   504	        destination: true,
   505	        pickupLatitude: true,
   506	        pickupLongitude: true,
   507	        destinationLatitude: true,
   508	        destinationLongitude: true,
   509	        scheduledDate: true,
   510	        estimatedFare: true,
   511	        bookingId: true,
   512	        agreedBidId: true,
   513	      },
   514	    });
   515	
   516	    if (!request) {
   517	      throw new Error("Marketplace request not found");
   518	    }
   519	
   520	    if (request.customerId !== customerId) {
   521	      throw new Error("Access denied");
   522	    }
   523	
   524	    if (request.status !== "OPEN") {
   525	      throw new Error("Marketplace request is no longer accepting bid selection");
   526	    }
   527	
   528	    /*
   529	     * Locate the requested bid and make sure it belongs to this request.
   530	     */
   531	    const bid = await tx.marketplaceBid.findFirst({
   532	      where: {
   533	        id: bidId,
   534	        requestId,
   535	      },
   536	      select: {
   537	        id: true,
   538	        requestId: true,
   539	        transporterId: true,
   540	        vehicleId: true,
   541	        amount: true,
   542	        message: true,
   543	        status: true,
   544	        expiresAt: true,
   545	      },
   546	    });
   547	
   548	    if (!bid) {
   549	      throw new Error("Marketplace bid not found");
   550	    }
   551	
   552	    if (bid.status !== "PENDING") {
   553	      throw new Error("Only pending bids can be selected");
   554	    }
   555	
   556	    if (bid.expiresAt && bid.expiresAt <= new Date()) {
   557	      throw new Error("Marketplace bid has expired");
   558	    }
   559	
   560	    /*
   561	     * Revalidate the transporter at the moment of acceptance.
   562	     * Visibility at bid-submission time is not sufficient because
   563	     * transporter status can change before the customer accepts.
   564	     */
   565	    const transporter = await tx.user.findUnique({
   566	      where: {
   567	        id: bid.transporterId,
   568	      },
   569	      select: {
   570	        id: true,
   571	        role: true,
   572	        status: true,
   573	        transporterTier: true,
   574	        transporterProfile: {
   575	          select: {
   576	            verificationStatus: true,
   577	          },
   578	        },
   579	      },
   580	    });
   581	
   582	    if (!transporter || transporter.role !== "TRANSPORTER") {
   583	      throw new Error("Selected transporter is invalid");
   584	    }
   585	
   586	    if (transporter.status !== "ACTIVE") {
   587	      throw new Error("Selected transporter account is not active");
   588	    }
   589	
   590	    if (
   591	      !transporter.transporterProfile ||
   592	      transporter.transporterProfile.verificationStatus !== "APPROVED"
   593	    ) {
   594	      throw new Error("Selected transporter is not approved");
   595	    }
   596	
   597	    /*
   598	     * Revalidate the vehicle and reserve it atomically.
   599	     * This prevents two concurrent operations from assigning the same
   600	     * available vehicle to separate trips.
   601	     */
   602	    const vehicle = await tx.vehicle.findUnique({
   603	      where: {
   604	        id: bid.vehicleId,
   605	      },
   606	      select: {
   607	        id: true,
   608	        transporterId: true,
   609	        vehicleType: true,
   610	        vehicleClass: true,
   611	        verificationStatus: true,
   612	        availabilityStatus: true,
   613	      },
   614	    });
   615	
   616	    if (!vehicle) {
   617	      throw new Error("Selected vehicle not found");
   618	    }
   619	
   620	    if (vehicle.transporterId !== bid.transporterId) {
   621	      throw new Error("Selected vehicle does not belong to transporter");
   622	    }
   623	
   624	    if (vehicle.verificationStatus !== "APPROVED") {
   625	      throw new Error("Selected vehicle is not approved");
   626	    }
   627	
   628	    if (vehicle.availabilityStatus !== "AVAILABLE") {
   629	      throw new Error("Selected vehicle is no longer available");
   630	    }
   631	
   632	    if (
   633	      request.truckCategory &&
   634	      vehicle.vehicleType !== request.truckCategory &&
   635	      vehicle.vehicleClass !== request.truckCategory
   636	    ) {
   637	      throw new Error("Selected vehicle no longer matches truck category");
   638	    }
   639	
   640	    /*
   641	     * Claim the marketplace request first.
   642	     *
   643	     * The conditional update is the concurrency guard:
   644	     * only one transaction can successfully move OPEN -> AGREED.
   645	     */
   646	    const claimedRequest = await tx.marketplaceRequest.updateMany({
   647	      where: {
   648	        id: requestId,
   649	        customerId,
   650	        status: "OPEN",
   651	      },
   652	      data: {
   653	        status: "AGREED",
   654	      },
   655	    });
   656	
   657	    if (claimedRequest.count !== 1) {
   658	      throw new Error("Marketplace request has already been processed");
   659	    }
   660	
   661	    /*
   662	     * Reserve the vehicle.
   663	     */
   664	    const reservedVehicle = await tx.vehicle.updateMany({
   665	      where: {
   666	        id: vehicle.id,
   667	        transporterId: bid.transporterId,
   668	        availabilityStatus: "AVAILABLE",
   669	        verificationStatus: "APPROVED",
   670	      },
   671	      data: {
   672	        availabilityStatus: "ON_TRIP",
   673	      },
   674	    });
   675	
   676	    if (reservedVehicle.count !== 1) {
   677	      throw new Error("Selected vehicle is no longer available");
   678	    }
   679	
   680	    /*
   681	     * Select exactly this bid.
   682	     */
   683	    const selectedAt = new Date();
   684	
   685	    const selectedBid = await tx.marketplaceBid.updateMany({
   686	      where: {
   687	        id: bid.id,
   688	        requestId,
   689	        status: "PENDING",
   690	      },
   691	      data: {
   692	        status: "SELECTED",
   693	        selectedAt,
   694	      },
   695	    });
   696	
   697	    if (selectedBid.count !== 1) {
   698	      throw new Error("Selected bid is no longer available");
   699	    }
   700	
   701	    /*
   702	     * All other pending bids lose the competition.
   703	     */
   704	    await tx.marketplaceBid.updateMany({
   705	      where: {
   706	        requestId,
   707	        id: {
   708	          not: bid.id,
   709	        },
   710	        status: "PENDING",
   711	      },

     8	function distanceKm(
     9	  lat1: number,
    10	  lon1: number,
    11	  lat2: number,
    12	  lon2: number,
    13	): number {
    14	  const toRadians = (value: number) =>
    15	    (value * Math.PI) / 180;
    16	
    17	  const dLat = toRadians(lat2 - lat1);
    18	  const dLon = toRadians(lon2 - lon1);
    19	
    20	  const a =
    21	    Math.sin(dLat / 2) ** 2 +
    22	    Math.cos(toRadians(lat1)) *
    23	      Math.cos(toRadians(lat2)) *
    24	      Math.sin(dLon / 2) ** 2;
    25	
    26	  const clampedA = Math.min(1, Math.max(0, a));
    27	
    28	  return (
    29	    EARTH_RADIUS_KM *
    30	    2 *
    31	    Math.atan2(
    32	      Math.sqrt(clampedA),
    33	      Math.sqrt(1 - clampedA),
    34	    )
    35	  );
    36	}
